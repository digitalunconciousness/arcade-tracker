# Database.
# SQLAlchemy is pinned deliberately. It is only a transitive dependency of
# Flask-SQLAlchemy, and leaving it unpinned resolves to 2.1.x, which changed the
# default driver for "postgresql://" URLs from psycopg2 to psycopg 3 -- so an
# unpinned install cannot reach the live database at all:
#   ModuleNotFoundError: No module named 'psycopg'
# 2.0.49 + psycopg2-binary is what the deployment runs today. Moving to
# SQLAlchemy 2.1 + psycopg 3 is a separate change, not a deployment detail.
SQLAlchemy==2.0.49
psycopg2-binary==2.9.12

# WSGI server (the Flask development server is not for production use)

# gpiozero is a SERVER dependency, not a Pi-only one, however much it looks like
# one. app/routes/skeeball.py imports gpio_init, which does `from gpiozero import
# Device` at module level and takes MockFactory from gpiozero for mock mode -- so
# the skeeball routes need it even on hardware-free x86. A static grep for
# "import gpiozero" under app/ finds nothing and is misleading; the import is
# transitive. RPi.GPIO is genuinely Pi-only and gpio_init already guards it.
gpiozero==2.0.1
pyserial==3.5
